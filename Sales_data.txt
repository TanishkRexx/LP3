
# Sales Data Clustering using K-Means
# Elbow Method

import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Step 1: Load dataset
data = pd.read_csv("sales_data_sample.csv", encoding="latin1")

print("Dataset Shape:", data.shape)

# Step 2: Select numerical columns
X = data[['QUANTITYORDERED', 'PRICEEACH', 'SALES']]

# Step 3: Handle missing values
X = X.fillna(X.median())

# Step 4: Feature scaling
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 5: Elbow Method
wcss = []

for i in range(1, 11):
    kmeans = KMeans(
        n_clusters=i,
        random_state=42,
        n_init=10
    )

    kmeans.fit(X_scaled)
    wcss.append(kmeans.inertia_)

# Step 6: Plot Elbow Graph
plt.plot(range(1, 11), wcss, marker='o')

plt.title("Elbow Method")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.show()

# Step 7: Apply K-Means
kmeans = KMeans(
    n_clusters=3,
    random_state=42,
    n_init=10
)

data['Cluster'] = kmeans.fit_predict(X_scaled)

# Step 8: Display results
print("\nCluster Results:")
print(data[['SALES', 'QUANTITYORDERED', 'Cluster']].head(10))

print("\nNumber of records in each cluster:")
print(data['Cluster'].value_counts())

# Step 9: Visualize clusters
plt.scatter(
    X_scaled[:, 0],
    X_scaled[:, 2],
    c=data['Cluster'],
    cmap='viridis'
)

plt.title("K-Means Sales Clustering")
plt.xlabel("Quantity Ordered (Scaled)")
plt.ylabel("Sales (Scaled)")
plt.show()
